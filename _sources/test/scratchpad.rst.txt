  public static final double windChillFactor(final double t, final double ws, char unit) {
    unit = normalizeUnit(unit);
    final double tf = (unit == 'f') ? t : Double.valueOf(t).toFahrenheit();
    Double wc = 0d;

    /** no wind chill for temperatures > 50F or wind speeds < 3mph */
    if (!(tf > 50 || ws < 3)) {
      final double ws_mph  = ws * 2.237;
      wc  = (35.74 + 0.6215 * tf - 35.75 * (Math.pow(ws_mph, 0.16)) + 0.4275 * tf * (Math.pow(ws_mph, 0.16)));
    } else {
      wc = t;
    }
    return (unit == 'c') ? wc.toCelsius().roundAt(2) : wc.roundAt(2);
  }
